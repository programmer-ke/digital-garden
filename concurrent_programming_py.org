** 1. Thread-Safe Counter

*** Prompt

Implement a class ~ThreadSafeCounter~ with methods ~increment()~ and ~get()~.
It must be safe to call ~increment()~ from many threads simultaneously.

*** Test

def test_thread_safe_counter():

    n_threads = 10
    n_increments = 1000

    def worker():

        for _ in range(n_increments):
            counter.increment()

    threads = [threading.Thread(target=worker) for _ in range(n_threads)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()

#+begin_src python :tangle ex.py
class ThreadSafeCounter:
    def __init__(self):
        self.count = 0
        self._lock = threading.Lock()

    def increment(self):
        with self._lock:
            self.count += 1

    def get(self):
        with self._lock:
            return self.count

def producer_consumer(items):

    results = []
    q = queue.Queue()

    def consumer():

        while True:
            item = q.get()
            if item is None:
                break
            results.append(item)

        results.sort()

    t = threading.Thread(target=consumer)
    t.start()

    for i in items:
        q.put(i)

    q.put(None)
    t.join()

    return results

def wait_for_event(seconds):

    event = threading.Event()

    def setter(timeout):
        time.sleep(timeout)
        event.set()

    t = threading.Thread(target=setter, args=(seconds,), daemon=True)
    t.start()

    return event.wait(seconds + 0.5)

#+begin_src python :tangle ex.py
def recepient(conn):

    while message := conn.recv():
        if message == "ping":
            conn.send("pong")
        else:
            conn.send("?")


def ping_pong(rounds):

    parent_conn, child_conn = multiprocessing.Pipe()
    p = multiprocessing.Process(target=recepient, args=(child_conn,))

    p.start()

    responses = []

    for _ in range(rounds):
        parent_conn.send("ping")
        responses.append(parent_conn.recv())

    parent_conn.send(None)
    p.join()
    return responses

#+begin_src python :tangle ex.py
def fetch_all(urls, max_workers, fetch_fn):
    results = {}
    with concurrent.futures.ThreadPoolExecutor(max_workers) as executor:
        future_to_url = {executor.submit(fetch_fn, url): url for url in urls}
        for future in concurrent.futures.as_completed(future_to_url):
            try:
                result = future.result()
            except ValueError:
                result = None

            url = future_to_url[future]
            results[url] = result
    return results

*** Implementation

#+begin_src python :tangle ex.py
class ReadWriteLock:
    """A lock only allows reads or one write exclusively (shared-exclusive)

    Implemented as a monitor over a mutex and two condition variables
    """

    def __init__(self):
        self.read_count = 0
        self.writer_active = False

        self.lock = threading.Lock()

        self.can_read = threading.Condition(self.lock)
        self.can_write = threading.Condition(self.lock)

    def acquire_read(self):
        """Add reader as long as writer is inactive"""

        with self.lock:
            while self.writer_active:
                self.can_read.wait()
            self.read_count += 1

    def release_read(self):
        """Remove reader and notify writer if non left"""

        with self.lock:
            self.read_count -= 1
            if self.read_count == 0:
                self.can_write.notify()

    def acquire_write(self):
        """Activate writer if there are no readers"""

        with self.lock:
            while self.read_count or self.writer_active:
                self.can_write.wait()
            self.writer_active = True

    def release_write(self):
        """Deactivate writer"""

        with self.lock:
            self.writer_active = False
            self.can_read.notify_all()
            self.can_write.notify()

** 8. Rate Limiter (Token Bucket)

*** Implementation

#+begin_src python :tangle ex.py
class TokenBucket:
    def __init__(self, rate, capacity):
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.last_refill = None
        self.lock = threading.Lock()

    def acquire(self, tokens=1):

        with self.lock:

            now = time.monotonic()

            if self.last_refill is not None:
                seconds_since = now - self.last_refill
                additional_tokens = seconds_since * self.rate
                self.tokens = min(self.capacity, self.tokens + additional_tokens)

            self.last_refill = now

            if self.tokens >= tokens:
                self.tokens -= tokens
                result = True
            else:
                result = False

        return result

** 9. Deadlock-Free Bank Transfer

*** Prompt

Implement ~transfer(from_acct: Account, to_acct: Account, amount: float) ->

def test_concurrent_transfers_no_deadlock():
    accounts = [Account(id=i, balance=1000.0) for i in range(10)]
    total_before = sum(a.balance for a in accounts)

    def worker():
        import random

        for _ in range(100):
            i, j = random.sample(range(10), 2)
            transfer(accounts[i], accounts[j], 1.0)

    threads = [threading.Thread(target=worker) for _ in range(10)]

*** Implementation

#+begin_src python :tangle ex.py
class Account:
    def __init__(self, id, balance):
        self.id = id
        self.balance = balance
        self.lock = threading.Lock()


def transfer(from_, to, amount):

    first, nxt = (from_, to) if from_.id < to.id else (to, from_)

    with first.lock, nxt.lock:

        if from_.balance < amount:
            raise ValueError(f"balance {from_.balance} cannot cover amount {amount}")

        from_.balance -= amount
        to.balance += amount

#+begin_src python :tangle ex.py
async def fetch_all_async(urls, max_concurrent, fetch_fn):

    semaphore = asyncio.Semaphore(max_concurrent)

    async def fetch(url):
        async with semaphore:
            try:
                return await fetch_fn(url)
            except ValueError:
                return None

    result = {}
    fetches = await asyncio.gather(
        *[fetch(url) for url in urls], return_exceptions=True
    )
    for i, fetch_result in enumerate(fetches):
        result[urls[i]] = fetch_result
    return result

#+begin_src python :tangle ex.py
def f_doubler(source, dest):
    for num in iter(source.get, None):
        dest.put(num * 2)
    dest.put(None)


def f_adder(source, dest):
    for num in iter(source.get, None):
        dest.put(num + 1)
    dest.put(None)


def run_pipeline(nums):

    q_input = multiprocessing.Queue()
    q_intmd = multiprocessing.Queue()
    q_result = multiprocessing.Queue()

    tasks = [(f_doubler, (q_input, q_intmd)), (f_adder, (q_intmd, q_result))]

    children = []
    for worker, args in tasks:
        children.append(multiprocessing.Process(target=worker, args=args))

    for p in children:
        p.start()

    for n in nums:
        q_input.put(n)
    q_input.put(None)

    results = list(sorted(iter(q_result.get, None)))

    for p in children:
        p.join()

    return results

class SimpleThreadPool:
    def __init__(self, num_workers):
        self._lock = threading.Lock()
        self.queue = queue.Queue()
        self.threads = [
            threading.Thread(target=self._thread, args=(self.queue,))
            for _ in range(num_workers)
        ]
        self._is_shutdown = False
        for t in self.threads:
            t.start()

    @staticmethod
    def _thread(queue):
        while task := queue.get():
            future, f, *args = task

            try:
                result = f(*args)
            except Exception as exc:
                future.set_exception(exc)
            else:
                future.set_result(result)
            queue.task_done()
        queue.task_done()

    def submit(self, func, *args):
        with self._lock:
            if self._is_shutdown:
                raise RuntimeError(
                    "Pool has been shutdown. Cannot accept additional tasks"
                )
            future = concurrent.futures.Future()
            self.queue.put((future, func, *args))
            return future

    def shutdown(self, wait=True):
        with self._lock:
            if self._is_shutdown:
                return

            self._is_shutdown = True

            for _ in range(len(self.threads)):
                self.queue.put(None)

        if wait:
            for t in self.threads:
                t.join()

    HEAD_SLOT = 0
    TAIL_SLOT = 1

        self._lock = multiprocessing.Lock()
        self._shm = shared_memory.SharedMemory(
            create=True, size=(size + self.HEADER_SIZE) * self.INT_SIZE
        )
        self._arr = self._shm.buf.cast("i")

        self._arr[self.COUNT_SLOT] = self._arr[self.TAIL_SLOT] = self._arr[
            self.HEAD_SLOT
        ] = 0

    def write(self, value):
        with self._lock:
            buffer_is_full = self._arr[self.COUNT_SLOT] == self._size
            if buffer_is_full:
                return False
            tail_idx = self._arr[self.TAIL_SLOT]
            self._arr[self.HEADER_SIZE + tail_idx] = value
            self._arr[self.COUNT_SLOT] += 1
            self._arr[self.TAIL_SLOT] = (tail_idx + 1) % self._size
        return True

    def read(self):
        with self._lock:
            if self._arr[self.COUNT_SLOT] == 0:
                return None
            head_idx = self._arr[self.HEAD_SLOT]
            value = self._arr[self.HEADER_SIZE + head_idx]
            self._arr[self.COUNT_SLOT] -= 1
            self._arr[self.HEAD_SLOT] = (head_idx + 1) % self._size
        return value

    def close(self):
        """Release local process' reference to shared memory"""
        self._arr.release()  # release the memory view, otherwise shm won't close
        self._shm.close()

    def unlink(self):
        """Delete the underlying shared memory"""
        self._shm.unlink()
#+end_src

async def run_blocking_in_thread(f, *args):
    result = await asyncio.to_thread(f, *args)
    return result


async def parallel_blocking(fns, max_concurrent):

    semaphore = asyncio.Semaphore(max_concurrent)

    async def runner(fn):
        async with semaphore:
            return await asyncio.to_thread(fn)

    return await asyncio.gather(*[runner(fn) for fn in fns])

** 20. Concurrent LRU Cache

*** Prompt

Implement a ~ConcurrentLRUCache~ class:
- ~__init__(self, capacity: int)~
- ~get(self, key) -> Any | None~
- ~put(self, key, value) -> None~

Must be thread-safe and evict the least-recently-used item when full. ~get~
counts as a "use."

*** Test

def test_thread_safety():
    cache = ConcurrentLRUCache(capacity=100)

    def worker(start):
        for i in range(50):
            key = f"key-{start}-{i}"
            cache.put(key, i)
            cache.get(key)

    threads = [threading.Thread(target=worker, args=(t,)) for t in range(10)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    # No crash = thread-safe; invariant maintained
    assert len(cache) == 100

#+begin_src python :tangle ex.py
class ConcurrentLRUCache:
    def __init__(self, capacity):
        self._capacity = capacity
        self._cache = collections.OrderedDict()
        self._lock = threading.Lock()

    def __len__(self):
        return len(self._cache)

    def get(self, key):
        with self._lock:
            if key in self._cache:
                self._cache.move_to_end(key)
                return self._cache[key]
            return None

    def put(self, key, value):
        with self._lock:
            try:
                self._cache.move_to_end(key)
            except KeyError:
                # key doesn't yet exist, skip
                pass

            self._cache[key] = value

            if len(self._cache) > self._capacity:
                self._cache.popitem(last=False)
