Write a decorator =@add_get_info= that adds a =get_info()= method to

        x = 1

    assert "bar" in info["methods"]
    assert info["attributes"]["x"] == 1

#+begin_src python
def add_get_info(cls):

    def get_info(self):

        name = type(self).__name__
        methods, attributes = [], {}

        for k, v in type(self).__dict__.items():
            if k.startswith("__"):
                continue
            if inspect.isfunction(v):
                methods.append(k)
            else:
                attributes[k] = v

        attributes |= vars(self)

        return {"name": name, "methods": methods, "attributes": attributes}

    cls.get_info = get_info
    return cls

class UpperMeta(type):
    def __new__(mcs, name, bases, namespace):

        non_dunder_methods = [
            (k, v)
            for k, v in namespace.items()
            if inspect.isfunction(v) and not k.startswith("__")

class Typed:
    def __init__(self, type_):
        self._type = type_

    def __set_name__(self, owner, name):
        self._storage_name = name

    def __get__(self, instance, owner):
        if instance is None:
            return self
        try:
            return instance.__dict__[self._storage_name]

            raise TypeError(f"expected type {self._type}, got {type(value)}")
        instance.__dict__[self._storage_name] = value

#+begin_src python
class PropMeta(type):
    def __new__(metacls, name, bases, namespace):
        attrs = [k for k in namespace if not k.startswith("__") and k.startswith("_")]

        def create_accessors(key):
            def fset(self, val):
                setattr(self, key, val)

            def fget(self):
                return getattr(self, key)

            return fget, fset

        for k in attrs:
            namespace[k[1:]] = property(*create_accessors(k))

        return super().__new__(metacls, name, bases, namespace)

Build a mini-ORM: a =Model= base class with a metaclass =ModelMeta=

class Field:
    def __init__(self, field_type):
        self.ftype = field_type

    def __set_name__(self, owner, name):
        self.storage_name = name


class ModelMeta(type):
    def __init__(cls, name, bases, namespace):
        super().__init__(name, bases, namespace)
        cls._fields = {}

        for base in reversed(bases):
            try:
                cls._fields |= base._fields
            except AttributeError:
                pass
        cls._fields |= {k: v for k, v in namespace.items() if isinstance(v, Field)}


class Model(metaclass=ModelMeta):

    @classmethod
    def create_table(cls):
        typed_columns = [(name, field.ftype) for name, field in cls._fields.items()]
        table_name = cls.__name__.lower()
        sql_columns = ",\n".join(
            f"{name} {field_type.__name__}" for name, field_type in typed_columns
        )
        return textwrap.dedent(
            f"""CREATE TABLE {table_name} (
            {sql_columns}
        );"""
        )

class Builder:
    def __init__(self, class_name):
        self.name = class_name
        self.attrs = {}
        self.methods = {}

    def add_attr(self, name, value):
        self.attrs[name] = value
        return self

    def add_method(self, name, func):
        self.methods[name] = func
        return self

    def build(self):
        namespace = self.attrs | self.methods
        return type(self.name, (), namespace)

class SingletonMeta(type):
    _instances = {}
    _lock = threading.Lock()

    def __call__(cls, *args, **kwargs):
        with cls._lock:
            if cls not in cls._instances:
                cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

class TimeMeta(type):

class AbstractMeta(type):
    def __init__(cls, name, bases, namespace):
        super().__init__(name, bases, namespace)

        cls._abstract_methods = set()

        for k, v in cls.__dict__.items():
            if getattr(v, "is_abstract", False):
                cls._abstract_methods.add(k)

        for base in cls.__mro__[1:]:
            if hasattr(base, "_abstract_methods"):
                cls._abstract_methods |= base._abstract_methods

    def __call__(cls, *args, **kwargs):
        for method in cls._abstract_methods:
            if (func := inspect.getattr_static(cls, method)) and getattr(
                func, "is_abstract", False
            ):
                raise TypeError(f"define concrete method: {method}")
        return super().__call__(*args, **kwargs)


def abstractmethod(func):
    func.is_abstract = True
    return func

#+begin_src python
class SerdeField:
    def __init__(self, ftype, optional=False):
        self._ftype = ftype
        self._optional = optional

    def __set_name__(self, owner, name):
        self._storage_name = name

    def __get__(self, instance, owner):
        if instance is None:
            return self
        try:
            return instance.__dict__[self._storage_name]
        except KeyError as exc:
            if self._optional:
                return None
            raise AttributeError(f"missing attribute: {self._storage_name}") from exc

    def __set__(self, instance, value):
        if not isinstance(value, self._ftype) and not (value is None and self._optional):
            raise TypeError(f"expected type {self._ftype}, got {type(value)}")
        instance.__dict__[self._storage_name] = value


class SerdeModel:

    def __init__(self, **kwargs):
        for k, v in kwargs.items():
            setattr(self, k, v)

    def to_dict(self):
        fields = {}
        for klass in type(self).__mro__:
            fields |= {
                k: v for k, v in vars(klass).items() if isinstance(v, SerdeField)
            }
        return {k: getattr(self, k) for k in fields}

    @classmethod
    def from_dict(cls, d):
        return cls(**d)
#+end_src
